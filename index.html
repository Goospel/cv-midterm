// ============================================================
// 03. 부분 영상(ROI) / 영상 반전 / 마스크 연산
//     (Chapter03 부분행렬, Chapter04 마스크)
// ============================================================
#include "opencv2/opencv.hpp"
#include <iostream>

using namespace std;
using namespace cv;

int main() {
	Mat img = imread("lenna.bmp");
	if (img.empty()) { cerr << "Image load failed!" << endl; return -1; }

	// ---------- ROI : 영상의 일부만 잘라내기 ----------
	// 잘라낸 결과는 원본을 "가리키는" 부분 행렬이다 (복사가 아니다)
	Mat roi = img(Rect(100, 100, 200, 150));   // Rect(x, y, width, height)

	// ROI 를 고치면 원본도 같이 바뀐다 -> ROI 만 반전시키는 방법
	roi = ~roi;                                 // 또는 bitwise_not(roi, roi);
	imshow("roi inversed", img);

	// 원본을 지키고 싶으면 clone() 으로 복사본을 만든 뒤 자른다
	Mat img2 = imread("lenna.bmp");
	Mat copy = img2.clone();
	Mat roi2 = copy(Rect(100, 100, 200, 150));
	roi2 = ~roi2;                               // img2 는 그대로, copy 만 바뀐다

	// 행 / 열 단위로 자르기
	// Mat row10 = img.row(10);
	// Mat col5  = img.col(5);
	// Mat rows  = img.rowRange(10, 20);
	// Mat cols  = img.colRange(5, 15);

	// ---------- 영상 전체 반전 ----------
	Mat inversed = ~img2;          // 255 - 픽셀값
	imshow("inversed", inversed);

	// ---------- 마스크 연산 ----------
	// mask 는 CV_8UC1 이고, 값이 0이 아닌 위치에만 복사된다
	Mat src = imread("airplane.bmp", IMREAD_COLOR);
	Mat mask = imread("mask_plane.bmp", IMREAD_GRAYSCALE);
	Mat dst = imread("field.bmp", IMREAD_COLOR);

	if (!src.empty() && !mask.empty() && !dst.empty()) {
		src.copyTo(dst, mask);     // 비행기 모양 부분만 들판 위에 옮겨 붙인다
		imshow("mask copyTo", dst);
	}

	// 마스크 위치에만 한 가지 색 칠하기
	Mat img3 = imread("lenna.bmp");
	Mat smile = imread("mask_smile.bmp", IMREAD_GRAYSCALE);
	if (!smile.empty()) {
		img3.setTo(Scalar(0, 255, 255), smile);   // 마스크 부분만 노랗게
		imshow("mask setTo", img3);
	}

	waitKey(0);
	destroyAllWindows();

	return 0;
}

// ============================================================
// 04. 픽셀 값 참조 / 채널 분리·합성
//     (Chapter03 행렬의 원소 값 참조)
// ============================================================
#include "opencv2/opencv.hpp"
#include <iostream>

using namespace std;
using namespace cv;

int main() {
	Mat gray = imread("lenna.bmp", IMREAD_GRAYSCALE);
	Mat img = imread("lenna.bmp", IMREAD_COLOR);
	if (gray.empty() || img.empty()) { cerr << "Image load failed!" << endl; return -1; }

	// ---------- at() 으로 픽셀 하나 읽고 쓰기 ----------
	// 첫 번째 인자가 "행 번호 = y", 두 번째가 "열 번호 = x" 이다. 순서 주의!
	uchar v = gray.at<uchar>(100, 50);           // (x=50, y=100) 위치의 밝기
	gray.at<uchar>(100, 50) = 255;

	Vec3b px = img.at<Vec3b>(100, 50);           // 컬러는 Vec3b
	int b = px[0];     // BLUE
	int g = px[1];     // GREEN
	int r = px[2];     // RED
	cout << "pixel B=" << b << " G=" << g << " R=" << r << endl;

	img.at<Vec3b>(100, 50)[0] = 0;               // BLUE  를 0 으로
	img.at<Vec3b>(100, 50)[1] = 255;             // GREEN 를 255 로
	img.at<Vec3b>(100, 50)[2] = 255;             // RED   를 255 로

	// ---------- 전체 픽셀 돌기 (가장 많이 쓰는 형태) ----------
	Mat dst = img.clone();
	for (int j = 0; j < dst.rows; j++) {         // j = y (세로)
		for (int i = 0; i < dst.cols; i++) {     // i = x (가로)
			dst.at<Vec3b>(j, i)[0] = 255 - img.at<Vec3b>(j, i)[0];
			dst.at<Vec3b>(j, i)[1] = 255 - img.at<Vec3b>(j, i)[1];
			dst.at<Vec3b>(j, i)[2] = 255 - img.at<Vec3b>(j, i)[2];
		}
	}
	imshow("inversed", dst);

	// 그레이스케일 전체 돌기
	Mat gdst = gray.clone();
	for (int j = 0; j < gdst.rows; j++)
		for (int i = 0; i < gdst.cols; i++)
			gdst.at<uchar>(j, i) = 255 - gray.at<uchar>(j, i);

	// ---------- ptr() 로 한 줄씩 접근 (at 보다 빠르다) ----------
	Mat pdst = gray.clone();
	for (int j = 0; j < pdst.rows; j++) {
		uchar* p = pdst.ptr<uchar>(j);           // j 번째 행의 첫 주소
		for (int i = 0; i < pdst.cols; i++)
			p[i] = 255 - p[i];
	}

	// ---------- 채널 분리 / 합성 ----------
	vector<Mat> planes;
	split(img, planes);        // planes[0]=B, planes[1]=G, planes[2]=R

	imshow("B plane", planes[0]);
	imshow("G plane", planes[1]);
	imshow("R plane", planes[2]);

	planes[0] = 0;             // 파란색 채널을 모두 0 으로
	Mat merged;
	merge(planes, merged);     // 다시 합치기
	imshow("B removed", merged);

	waitKey(0);
	destroyAllWindows();

	return 0;
}

// ------------------------------------------------------------
// (1) 기본적인 방법 - 그냥 곱하기 (코드 5-5)
//     dst(x, y) = saturate( s * src(x, y) )
//     s > 1 이면 명암비 증가, 0 < s < 1 이면 감소
//     단점 : 전체가 밝아져서 밝은 쪽이 다 하얗게 날아간다
// ------------------------------------------------------------
void contrast1() {
	Mat src = imread("lenna.bmp", IMREAD_GRAYSCALE);
	if (src.empty()) { cerr << "Image load failed!" << endl; return; }

	float s = 2.f;
	Mat dst = s * src;

	imshow("src", src);
	imshow("dst", dst);
	waitKey();
	destroyAllWindows();

//     dst(x, y) = saturate( src(x, y) + (src(x, y) - 128) * alpha )
//
//     항상 (128, 128) 을 지나는 직선이라, 밝은 픽셀은 더 밝게
//     어두운 픽셀은 더 어둡게 되어 전체 밝기는 유지된다
//       alpha > 0        -> 명암비 증가
//       -1 <= alpha <= 0 -> 명암비 감소
// ------------------------------------------------------------
void contrast2() {
	Mat src = imread("lenna.bmp", IMREAD_GRAYSCALE);
	if (src.empty()) { cerr << "Image load failed!" << endl; return; }

	float alpha = 1.f;
	Mat dst = src + (src - 128) * alpha;

	imshow("dst", dst);

Mat calcGrayHist(const Mat& img) {
	CV_Assert(img.type() == CV_8UC1);

	Mat hist;
	int channels[] = { 0 };            // 0번 채널 (그레이스케일은 하나뿐)
	int dims = 1;                      // 1차원 히스토그램
	const int histSize[] = { 256 };    // 빈(bin) 개수 256개
	float graylevel[] = { 0, 256 };    // 값의 범위 0 이상 256 미만
	const float* ranges[] = { graylevel };

	calcHist(&img, 1, channels, noArray(), hist, dims, histSize, ranges);

	return hist;    // CV_32FC1, 원소 256개
}

// ------------------------------------------------------------
// (2) 히스토그램 그래프 그리기 (코드 5-8)
//
//  ※ 교재는 hist.at<float>(i, 0) 과 CV_Assert(hist.size() == Size(1, 256))
//     을 쓰는데, OpenCV 5.x 에서는 결과가 1 x 256 모양이라
//     그대로 쓰면 Assert 에서 멈추거나 값이 전부 0 으로 읽힌다.
//     아래처럼 hist.at<float>(i) 로 쓰면 4.x / 5.x 양쪽에서 다 맞는다.
// ------------------------------------------------------------
Mat getGrayHistImage(const Mat& hist) {
	CV_Assert(hist.type() == CV_32FC1);
	CV_Assert(hist.total() == 256);

	double histMax;
	minMaxLoc(hist, 0, &histMax);      // 가장 큰 빈도수를 찾는다

	Mat imgHist(100, 256, CV_8UC1, Scalar(255));   // 흰 바탕 256 x 100
	for (int i = 0; i < 256; i++) {
		line(imgHist, Point(i, 100),
			Point(i, 100 - cvRound(hist.at<float>(i) * 100 / histMax)), Scalar(0));
	}

	return imgHist;
}

// ------------------------------------------------------------
// (3) 히스토그램 스트레칭 (코드 5-9)
//
//     dst(x, y) = (src(x, y) - Gmin) * 255 / (Gmax - Gmin)
//
//     영상이 쓰는 밝기 구간을 0~255 전체로 쭉 늘려 명암비를 높인다.
//     OpenCV 에 전용 함수가 없어서 직접 쓴다.
// ------------------------------------------------------------
void histogram_stretching() {
	Mat src = imread("hawkes.bmp", IMREAD_GRAYSCALE);
	if (src.empty()) { cerr << "Image load failed!" << endl; return; }

	double gmin, gmax;
	minMaxLoc(src, &gmin, &gmax);      // 최솟값 / 최댓값 찾기

	Mat dst = (src - gmin) * 255 / (gmax - gmin);

	imshow("src", src);
	imshow("srcHist", getGrayHistImage(calcGrayHist(src)));
	imshow("dst", dst);
	imshow("dstHist", getGrayHistImage(calcGrayHist(dst)));

	waitKey();
	destroyAllWindows();
}

// ------------------------------------------------------------
// (4) Ends-In Search 스트레칭
//
//     하한(low) 아래는 전부 0, 상한(high) 위는 전부 255 로 만들고
//     그 사이만 늘린다. 아주 어둡거나 밝은 소수의 픽셀 때문에
//     스트레칭 효과가 안 나는 경우에 쓴다.
// ------------------------------------------------------------
void histogram_ends_in(int low = 40, int high = 200) {
	Mat src = imread("hawkes.bmp", IMREAD_GRAYSCALE);
	if (src.empty()) { cerr << "Image load failed!" << endl; return; }

	Mat dst(src.rows, src.cols, src.type());

	for (int j = 0; j < src.rows; j++) {
		for (int i = 0; i < src.cols; i++) {
			int v = src.at<uchar>(j, i);

			if (v <= low)
				dst.at<uchar>(j, i) = 0;
			else if (v >= high)
				dst.at<uchar>(j, i) = 255;
			else
				dst.at<uchar>(j, i) = saturate_cast<uchar>((v - low) * 255 / (high - low));
		}
	}

	imshow("src", src);
	imshow("srcHist", getGrayHistImage(calcGrayHist(src)));
	imshow("dst", dst);
	imshow("dstHist", getGrayHistImage(calcGrayHist(dst)));

	waitKey();
	destroyAllWindows();
}

이론 요약 (Chapter 01~03)

영상의 구조

  • 픽셀(pixel) — 영상을 이루는 최소 단위. 좌표는 0부터 시작(0-기반).
  • 행렬로 쓸 때 a(j, i) 에서 j = 행 = y, i = 열 = x. 그래서 OpenCV 도 at<uchar>(j, i) 가 (y, x) 순서다. 가장 많이 틀리는 곳.
  • 그레이스케일 — 밝기만, 0(검정) ~ 255(흰색). 1바이트 unsigned char = Windows 의 BYTE = 리눅스의 uint8_t = OpenCV 의 uchar.
  • 컬러(트루컬러) — R, G, B 각각 0~255. 한 픽셀에 3바이트. OpenCV 의 메모리 순서는 BGR 이라 Vec3b[0] 이 B, [2] 가 R.

자료형 이름

타입뜻at 에 쓰는 자료형
CV_8UC18비트 unsigned, 1채널 (그레이스케일)at<uchar>
CV_8UC38비트 unsigned, 3채널 (컬러)at<Vec3b>
CV_32FC132비트 float, 1채널 (히스토그램 등)at<float>

클래스멤버메모
